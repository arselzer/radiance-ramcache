cmake_minimum_required(VERSION 3.20)
project(glm53_cache_prototype LANGUAGES CXX)
find_package(Threads REQUIRED)
include(CTest)
add_library(expert_cache STATIC expert_cache.cpp)
target_include_directories(expert_cache PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(expert_cache PUBLIC cxx_std_20)
target_link_libraries(expert_cache PUBLIC Threads::Threads)
target_compile_options(expert_cache PRIVATE -Wall -Wextra -Wpedantic)
add_executable(expert_cache_test expert_cache_test.cpp)
target_link_libraries(expert_cache_test PRIVATE expert_cache)
add_test(NAME expert_cache_test COMMAND expert_cache_test)
set_tests_properties(expert_cache_test PROPERTIES TIMEOUT 30)

# Separate provider/trace source ownership; these reuse a vendored metadata-only JSON parser.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/file_provider.cpp")
  add_library(file_provider STATIC file_provider.cpp)
  target_include_directories(file_provider PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../radiance-src/vendor")
  target_link_libraries(file_provider PUBLIC expert_cache)
  target_compile_options(file_provider PRIVATE -Wall -Wextra -Wpedantic)
  foreach(tool trace_replay provider_trace_test)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${tool}.cpp")
      add_executable(${tool} "${tool}.cpp")
      target_link_libraries(${tool} PRIVATE file_provider)
    endif()
  endforeach()
  if(TARGET provider_trace_test)
    add_test(NAME provider_trace_test COMMAND provider_trace_test)
    set_tests_properties(provider_trace_test PROPERTIES TIMEOUT 30)
  endif()
endif()
