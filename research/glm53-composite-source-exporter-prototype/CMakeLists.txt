cmake_minimum_required(VERSION 3.24)
project(glm53_composite_source_exporter LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Prototype requires Linux renameat2 NOREPLACE; no overwrite fallback")
endif()
find_package(Threads REQUIRED)
set(RESEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RADIANCE_SOURCE "${RESEARCH_ROOT}/radiance-src" CACHE PATH "Pinned read-only source")
set(RADIANCE_HOST_BUILD "${RADIANCE_SOURCE}/build-iq2-host" CACHE PATH "Matching existing HOST Core archives; parent selects/validates")
if(NOT EXISTS "${RADIANCE_HOST_BUILD}/core/librad_core.a")
  message(FATAL_ERROR "Pass RADIANCE_HOST_BUILD pointing to matching existing HOST Core, no implicit Core build")
endif()
file(STRINGS "${RADIANCE_HOST_BUILD}/CMakeCache.txt" CORE_HOME REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" CORE_HOME "${CORE_HOME}")
file(REAL_PATH "${CORE_HOME}" CORE_HOME_REAL)
file(REAL_PATH "${RADIANCE_SOURCE}" EXPECTED_CORE_REAL)
if(NOT CORE_HOME_REAL STREQUAL EXPECTED_CORE_REAL)
  message(FATAL_ERROR "Program/Plan headers must match the selected Core archive build source")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/CORE_SOURCE_PINS.json" SOURCE_PINS)
string(JSON PIN_COUNT LENGTH "${SOURCE_PINS}")
math(EXPR PIN_LAST "${PIN_COUNT}-1")
foreach(I RANGE 0 ${PIN_LAST})
  string(JSON REL MEMBER "${SOURCE_PINS}" ${I})
  string(JSON EXPECTED GET "${SOURCE_PINS}" "${REL}")
  file(SHA256 "${RADIANCE_SOURCE}/${REL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Pinned source/private ABI mismatch: ${REL}")
  endif()
endforeach()
add_library(glm53_composite_exporter STATIC native_record_store.cpp)
target_compile_features(glm53_composite_exporter PUBLIC cxx_std_20)
target_include_directories(glm53_composite_exporter PUBLIC "${RADIANCE_SOURCE}/core" "${RADIANCE_SOURCE}/abi" "${RADIANCE_SOURCE}/vendor" "${RADIANCE_HOST_BUILD}/rad_include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(glm53_composite_exporter PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
add_executable(glm53_composite_exporter_test composite_exporter_test.cpp)
target_compile_features(glm53_composite_exporter_test PRIVATE cxx_std_20)
target_compile_options(glm53_composite_exporter_test PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_include_directories(glm53_composite_exporter_test PRIVATE "${RADIANCE_SOURCE}/tools")
target_link_libraries(glm53_composite_exporter_test PRIVATE glm53_composite_exporter
  "${RADIANCE_HOST_BUILD}/core/librad_core.a"
  "${RADIANCE_HOST_BUILD}/vendor/librad_vendor_llama.a"
  "${RADIANCE_HOST_BUILD}/vendor/librad_vendor_httplib.a" dl Threads::Threads)
# Parent invokes this tiny CPU-only test explicitly. No GPU/model/serving/install target.

target_link_options(glm53_composite_exporter_test PRIVATE "LINKER:--wrap=pread" "LINKER:--wrap=pwrite")
