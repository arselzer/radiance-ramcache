cmake_minimum_required(VERSION 3.24)
project(glm53_offline_synthetic_record_store LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Prototype requires Linux renameat2 NOREPLACE; no overwrite fallback")
endif()
find_package(Threads REQUIRED)
set(RESEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RADIANCE_SOURCE "${RESEARCH_ROOT}/radiance-src" CACHE PATH "Pinned read-only source")
set(RADIANCE_HOST_BUILD "${RADIANCE_SOURCE}/build-iq2-host" CACHE PATH "Matching existing HOST Core archives; parent selects/validates")
if(NOT EXISTS "${RADIANCE_HOST_BUILD}/core/librad_core.a")
  message(FATAL_ERROR "Pass RADIANCE_HOST_BUILD pointing to matching existing HOST Core, no implicit Core build")
endif()
add_library(glm53_native_record_store STATIC native_record_store.cpp)
target_compile_features(glm53_native_record_store PUBLIC cxx_std_20)
target_include_directories(glm53_native_record_store PUBLIC "${RADIANCE_SOURCE}/core" "${RADIANCE_SOURCE}/abi" "${RADIANCE_SOURCE}/vendor" "${RADIANCE_HOST_BUILD}/rad_include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(glm53_native_record_store PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
add_executable(glm53_native_record_store_test record_store_test.cpp)
target_compile_features(glm53_native_record_store_test PRIVATE cxx_std_20)
target_compile_options(glm53_native_record_store_test PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_include_directories(glm53_native_record_store_test PRIVATE "${RADIANCE_SOURCE}/tools")
target_link_libraries(glm53_native_record_store_test PRIVATE glm53_native_record_store
  "${RADIANCE_HOST_BUILD}/core/librad_core.a"
  "${RADIANCE_HOST_BUILD}/vendor/librad_vendor_llama.a"
  "${RADIANCE_HOST_BUILD}/vendor/librad_vendor_httplib.a" dl Threads::Threads)
# Parent invokes this tiny CPU-only test explicitly. No GPU/model/serving/install target.
