cmake_minimum_required(VERSION 3.24)
project(glm53_native_device_owner LANGUAGES CXX HIP)
find_package(Threads REQUIRED)
set(RESEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(RADIANCE_SOURCE "${RESEARCH_ROOT}/radiance-glm-main-tp4-cache-src" CACHE PATH "Matching isolated TinyAqlV1 Core source")
set(RADIANCE_BUILD "${RESEARCH_ROOT}/build-glm-engine-native-transport-aql" CACHE PATH "Matching whole HIP/AQL Core archive build")
set(PROVIDER_SOURCE "${RESEARCH_ROOT}/glm53-sharded-native-provider-prototype")
if(NOT EXISTS "${RADIANCE_BUILD}/core/librad_core.a" OR NOT EXISTS "${RADIANCE_SOURCE}/core/place/native_device_profile.h")
  message(FATAL_ERROR "Parent must first build the matching isolated TinyAqlV1 Core; no implicit Core build")
endif()
file(STRINGS "${RADIANCE_BUILD}/CMakeCache.txt" CORE_HOME REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" CORE_HOME "${CORE_HOME}")
file(REAL_PATH "${CORE_HOME}" CORE_HOME_REAL)
file(REAL_PATH "${RADIANCE_SOURCE}" EXPECTED_CORE_REAL)
if(NOT CORE_HOME_REAL STREQUAL EXPECTED_CORE_REAL)
  message(FATAL_ERROR "Private Core/Program/Plan/NativeInventory headers and archive source mismatch")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/CORE_SOURCE_PINS.json" CORE_PINS)
string(JSON PIN_COUNT LENGTH "${CORE_PINS}")
math(EXPR PIN_LAST "${PIN_COUNT}-1")
foreach(I RANGE 0 ${PIN_LAST})
  string(JSON REL MEMBER "${CORE_PINS}" ${I})
  string(JSON EXPECTED GET "${CORE_PINS}" "${REL}")
  file(SHA256 "${RADIANCE_SOURCE}/${REL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "TinyAqlV1 Core source mismatch: ${REL}")
  endif()
endforeach()
find_library(GLM53_HIP_RUNTIME_LIBRARY amdhip64 REQUIRED)
find_library(GLM53_HSA_RUNTIME_LIBRARY hsa-runtime64 REQUIRED)
set(GLM53_FROZEN_DEVICE_MODULE "" CACHE FILEPATH "Optional exact frozen six-row module, no HIP recompile")
set(GLM53_FROZEN_DEVICE_MODULE_SHA256 "7aeadd0ff26820c91c7a355de697d832b374a8d8e4379c7da94e368dbb9778f2")
if(GLM53_FROZEN_DEVICE_MODULE)
  file(SHA256 "${GLM53_FROZEN_DEVICE_MODULE}" MODULE_ACTUAL)
  if(NOT MODULE_ACTUAL STREQUAL GLM53_FROZEN_DEVICE_MODULE_SHA256)
    message(FATAL_ERROR "Frozen original DEVICE module hash mismatch")
  endif()
  add_library(glm53_native_device_rows SHARED IMPORTED GLOBAL)
  set_target_properties(glm53_native_device_rows PROPERTIES IMPORTED_LOCATION "${GLM53_FROZEN_DEVICE_MODULE}")
else()
add_library(glm53_native_device_rows SHARED reader_rows.cpp gemm_cfg.cpp
  "${RADIANCE_SOURCE}/libr4d/r4d_moe_mxfp4.hip"
  "${RADIANCE_SOURCE}/libr4d/r4d_moe.hip"
  "${RADIANCE_SOURCE}/libr4d/r4d_glm.hip"
  "${RADIANCE_SOURCE}/libr4d/r4d_layout.cpp"
  "${RADIANCE_SOURCE}/libr4d/r4d_fp8_layout.cpp"
  "${RADIANCE_SOURCE}/libr4d/r4d_shapes.cpp")
target_compile_features(glm53_native_device_rows PRIVATE cxx_std_20)
target_include_directories(glm53_native_device_rows PRIVATE "${RADIANCE_SOURCE}/abi" "${RADIANCE_SOURCE}/libr4d")
target_compile_options(glm53_native_device_rows PRIVATE -fno-fast-math -ffp-contract=off)
target_link_options(glm53_native_device_rows PRIVATE "LINKER:--no-undefined")
target_link_libraries(glm53_native_device_rows PRIVATE ${GLM53_HIP_RUNTIME_LIBRARY})
set_target_properties(glm53_native_device_rows PROPERTIES HIP_ARCHITECTURES gfx1201 HIP_STANDARD 20 LINKER_LANGUAGE CXX LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/research-kernels")
endif()
add_library(glm53_native_device_source STATIC native_source.cpp rank_native_store.cpp "${PROVIDER_SOURCE}/candidate_native_record_store.cpp")
target_compile_features(glm53_native_device_source PUBLIC cxx_std_20)
target_include_directories(glm53_native_device_source PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROVIDER_SOURCE}" "${RADIANCE_SOURCE}/core" "${RADIANCE_SOURCE}/abi" "${RADIANCE_SOURCE}/vendor" "${RADIANCE_BUILD}/rad_include")
target_compile_options(glm53_native_device_source PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
add_executable(glm53_native_device_owner_test native_device_owner_test.cpp)
target_compile_features(glm53_native_device_owner_test PRIVATE cxx_std_20)
target_compile_options(glm53_native_device_owner_test PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_include_directories(glm53_native_device_owner_test PRIVATE "${RADIANCE_SOURCE}/arch/common" "${RADIANCE_SOURCE}/tools")
target_link_libraries(glm53_native_device_owner_test PRIVATE glm53_native_device_source stdc++ "-Wl,--whole-archive" "${RADIANCE_BUILD}/core/librad_core.a" "-Wl,--no-whole-archive" "${RADIANCE_BUILD}/vendor/librad_vendor_llama.a" "${RADIANCE_BUILD}/vendor/librad_vendor_httplib.a" ${GLM53_HIP_RUNTIME_LIBRARY} ${GLM53_HSA_RUNTIME_LIBRARY} dl Threads::Threads)
target_link_options(glm53_native_device_owner_test PRIVATE "LINKER:--export-dynamic" "LINKER:--wrap=pread" "LINKER:--wrap=rad_memcpy_async" "LINKER:--wrap=rad_event_record" "LINKER:--wrap=rad_event_wait" "LINKER:--wrap=rad_stream_sync")
set_target_properties(glm53_native_device_owner_test PROPERTIES LINKER_LANGUAGE CXX)
add_dependencies(glm53_native_device_owner_test glm53_native_device_rows)
