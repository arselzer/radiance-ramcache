cmake_minimum_required(VERSION 3.24)
project(glm53_main_verified_components LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Actual pinned source/profile requires Linux")
endif()
find_package(Threads REQUIRED)
set(RESEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RADIANCE_SOURCE "${RESEARCH_ROOT}/radiance-glm-main-verified-source-v5-src" CACHE PATH "Frozen matching NEW whole Core")
set(RADIANCE_BUILD "" CACHE PATH "Parent-built matching whole Core")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/CORE_SOURCE_PINS.json")
  message(FATAL_ERROR "Component package awaits whole-Core source freeze/pins; do not configure mutable sources")
endif()
if(NOT EXISTS "${RADIANCE_BUILD}/core/librad_core.a")
  message(FATAL_ERROR "Supply matching parent-built whole Core; no implicit build")
endif()
function(core_cache KEY OUT)
  file(STRINGS "${RADIANCE_BUILD}/CMakeCache.txt" VALUE REGEX "^${KEY}:[^=]*=")
  string(REGEX REPLACE "^[^=]*=" "" VALUE "${VALUE}")
  set(${OUT} "${VALUE}" PARENT_SCOPE)
endfunction()
core_cache(CMAKE_HOME_DIRECTORY CORE_HOME)
file(REAL_PATH "${CORE_HOME}" ACTUAL_HOME)
file(REAL_PATH "${RADIANCE_SOURCE}" EXPECTED_HOME)
if(NOT ACTUAL_HOME STREQUAL EXPECTED_HOME)
  message(FATAL_ERROR "Private Core headers/archive source mismatch")
endif()
core_cache(CMAKE_CXX_COMPILER CORE_CXX)
file(REAL_PATH "${CORE_CXX}" CORE_CXX_REAL)
file(REAL_PATH "${CMAKE_CXX_COMPILER}" CHECK_CXX_REAL)
if(NOT CORE_CXX_REAL STREQUAL CHECK_CXX_REAL)
  message(FATAL_ERROR "Use the exact whole-Core C++ compiler")
endif()
core_cache(CMAKE_CXX_FLAGS CORE_FLAGS)
if(NOT CORE_FLAGS STREQUAL "" OR NOT CMAKE_CXX_FLAGS STREQUAL "")
  message(FATAL_ERROR "Unsupported global C++/ABI flags; use the pinned standard profile")
endif()
core_cache(CMAKE_BUILD_TYPE CORE_TYPE)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE "${CORE_TYPE}" CACHE STRING "Match whole Core" FORCE)
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL CORE_TYPE OR NOT CORE_TYPE MATCHES "^(Release|Debug)$")
  message(FATAL_ERROR "Require matching standard Release/Debug whole-Core profile")
endif()
string(TOUPPER "${CORE_TYPE}" ACTIVE_TYPE)
core_cache("CMAKE_CXX_FLAGS_${ACTIVE_TYPE}" CORE_ACTIVE_FLAGS)
if(CORE_TYPE STREQUAL "Release")
  set(EXPECTED_FLAGS "-O3 -DNDEBUG")
else()
  set(EXPECTED_FLAGS "-g")
endif()
if(NOT CORE_ACTIVE_FLAGS STREQUAL EXPECTED_FLAGS OR NOT "${CMAKE_CXX_FLAGS_${ACTIVE_TYPE}}" STREQUAL EXPECTED_FLAGS)
  message(FATAL_ERROR "Unsupported active compiler/ABI flags")
endif()
core_cache(RAD_ASAN CORE_ASAN)
core_cache(RAD_TSAN CORE_TSAN)
if(CORE_ASAN STREQUAL "ON")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
elseif(CORE_TSAN STREQUAL "ON")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/CORE_SOURCE_PINS.json" PINS)
string(JSON COUNT LENGTH "${PINS}")
math(EXPR LAST "${COUNT}-1")
foreach(I RANGE 0 ${LAST})
  string(JSON REL MEMBER "${PINS}" ${I})
  string(JSON EXPECTED GET "${PINS}" "${REL}")
  file(SHA256 "${RADIANCE_SOURCE}/${REL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Frozen Core source mismatch: ${REL}")
  endif()
endforeach()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/HELPER_SOURCE_PINS.json" HELPER_PINS)
string(JSON HELPER_COUNT LENGTH "${HELPER_PINS}")
math(EXPR HELPER_LAST "${HELPER_COUNT}-1")
foreach(I RANGE 0 ${HELPER_LAST})
  string(JSON REL MEMBER "${HELPER_PINS}" ${I})
  string(JSON EXPECTED GET "${HELPER_PINS}" "${REL}")
  file(SHA256 "${RESEARCH_ROOT}/${REL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Frozen reused helper mismatch: ${REL}")
  endif()
endforeach()
core_cache(RAD_WITH_HIP CORE_HIP)
option(RADIANCE_AQL_CORE "Explicit matching AQL archive; still no automatic execution" OFF)
if(CORE_HIP STREQUAL "OFF" AND RADIANCE_AQL_CORE)
  message(FATAL_ERROR "HOST Core cannot claim AQL profile")
elseif(NOT CORE_HIP STREQUAL "OFF" AND NOT RADIANCE_AQL_CORE)
  message(FATAL_ERROR "Explicit -DRADIANCE_AQL_CORE=ON needed for non-HOST Core")
endif()
if(RADIANCE_AQL_CORE)
  find_library(COMPONENT_HIP_LIBRARY amdhip64 REQUIRED)
  find_library(COMPONENT_HSA_LIBRARY hsa-runtime64 REQUIRED)
endif()
function(component TARGET SOURCE)
  add_executable(${TARGET} ${SOURCE})
  target_compile_features(${TARGET} PRIVATE cxx_std_20)
  target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  target_include_directories(${TARGET} PRIVATE "${RADIANCE_SOURCE}/core" "${RADIANCE_SOURCE}/abi"
    "${RADIANCE_SOURCE}/vendor" "${RADIANCE_SOURCE}/tools" "${RADIANCE_SOURCE}/arch/common" "${RADIANCE_BUILD}/rad_include")
  target_link_libraries(${TARGET} PRIVATE "-Wl,--whole-archive" "${RADIANCE_BUILD}/core/librad_core.a" "-Wl,--no-whole-archive"
    "${RADIANCE_BUILD}/vendor/librad_vendor_llama.a" "${RADIANCE_BUILD}/vendor/librad_vendor_httplib.a" Threads::Threads ${CMAKE_DL_LIBS})
  if(RADIANCE_AQL_CORE)
    target_compile_definitions(${TARGET} PRIVATE RAD_HAVE_HIP=1)
    target_link_libraries(${TARGET} PRIVATE ${COMPONENT_HIP_LIBRARY} ${COMPONENT_HSA_LIBRARY})
  endif()
  target_link_options(${TARGET} PRIVATE "LINKER:--export-dynamic" "LINKER:--wrap=rad_dev_alloc" "LINKER:--wrap=pread")
endfunction()
