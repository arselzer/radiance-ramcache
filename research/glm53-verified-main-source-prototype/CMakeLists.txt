cmake_minimum_required(VERSION 3.24)
project(glm53_verified_main_source LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Pinned real bundle/proof and POSIX generation checks require Linux")
endif()
find_package(Threads REQUIRED)
set(RESEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RADIANCE_SOURCE "${RESEARCH_ROOT}/radiance-glm-main-tp4-cache-src" CACHE PATH "Exact frozen matching research Core")
set(RADIANCE_BUILD "${RESEARCH_ROOT}/build-glm-main-tp4-cache-host-release" CACHE PATH "Parent-built matching whole HOST or AQL Core")
if(NOT EXISTS "${RADIANCE_BUILD}/core/librad_core.a")
  message(FATAL_ERROR "Supply a matching parent-built whole Core; no implicit Core build")
endif()
file(STRINGS "${RADIANCE_BUILD}/CMakeCache.txt" CORE_HOME REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" CORE_HOME "${CORE_HOME}")
file(REAL_PATH "${CORE_HOME}" ACTUAL_HOME)
file(REAL_PATH "${RADIANCE_SOURCE}" EXPECTED_HOME)
if(NOT ACTUAL_HOME STREQUAL EXPECTED_HOME)
  message(FATAL_ERROR "Private Program/Plan/header/archive source mismatch")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/CORE_SOURCE_PINS.json" PINS)
string(JSON COUNT LENGTH "${PINS}")
math(EXPR LAST "${COUNT}-1")
foreach(I RANGE 0 ${LAST})
  string(JSON REL MEMBER "${PINS}" ${I})
  string(JSON EXPECTED GET "${PINS}" "${REL}")
  file(SHA256 "${RADIANCE_SOURCE}/${REL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Frozen matching source mismatch: ${REL}")
  endif()
endforeach()
option(RADIANCE_AQL_CORE "Matching Core was built with HIP/AQL (no execution implied)" OFF)
file(STRINGS "${RADIANCE_BUILD}/CMakeCache.txt" HOST_PROFILE REGEX "^RAD_WITH_HIP:STRING=OFF$")
if(HOST_PROFILE AND RADIANCE_AQL_CORE)
  message(FATAL_ERROR "AQL fixture profile cannot consume a HOST Core")
elseif(NOT HOST_PROFILE AND NOT RADIANCE_AQL_CORE)
  message(FATAL_ERROR "Explicit -DRADIANCE_AQL_CORE=ON required for a non-HOST Core")
endif()
add_library(glm53_verified_main_source STATIC main_bundle_reader.cpp geometry/main_metadata.cpp
  storage/candidate_native_record_store.cpp storage/rank_native_store.cpp native_source.cpp)
target_compile_features(glm53_verified_main_source PUBLIC cxx_std_20)
target_include_directories(glm53_verified_main_source PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/storage"
  "${CMAKE_CURRENT_SOURCE_DIR}/producer" "${CMAKE_CURRENT_SOURCE_DIR}/geometry"
  "${RADIANCE_SOURCE}/core" "${RADIANCE_SOURCE}/abi" "${RADIANCE_SOURCE}/vendor" "${RADIANCE_BUILD}/rad_include")
target_compile_options(glm53_verified_main_source PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_link_libraries(glm53_verified_main_source PUBLIC Threads::Threads)
if(RADIANCE_AQL_CORE)
  find_library(MAIN_HIP_LIBRARY amdhip64 REQUIRED)
  find_library(MAIN_HSA_LIBRARY hsa-runtime64 REQUIRED)
  target_compile_definitions(glm53_verified_main_source PUBLIC RAD_HAVE_HIP=1)
  target_link_libraries(glm53_verified_main_source PUBLIC ${MAIN_HIP_LIBRARY} ${MAIN_HSA_LIBRARY})
endif()
function(main_source_check TARGET SOURCE)
  add_executable(${TARGET} ${SOURCE})
  target_compile_features(${TARGET} PRIVATE cxx_std_20)
  target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  target_include_directories(${TARGET} PRIVATE "${RADIANCE_SOURCE}/tools" "${RADIANCE_SOURCE}/arch/common")
  target_link_libraries(${TARGET} PRIVATE glm53_verified_main_source
    "-Wl,--whole-archive" "${RADIANCE_BUILD}/core/librad_core.a" "-Wl,--no-whole-archive"
    "${RADIANCE_BUILD}/vendor/librad_vendor_llama.a" "${RADIANCE_BUILD}/vendor/librad_vendor_httplib.a" Threads::Threads ${CMAKE_DL_LIBS})
  target_link_options(${TARGET} PRIVATE "LINKER:--export-dynamic")
endfunction()
main_source_check(glm53_verified_main_reader_refusals reader_refusal_check.cpp)
target_link_options(glm53_verified_main_reader_refusals PRIVATE "LINKER:--wrap=open" "LINKER:--wrap=openat" "LINKER:--wrap=pread" "LINKER:--wrap=rad_dev_alloc")
main_source_check(glm53_verified_main_generic_regression storage/sharded_native_store_test.cpp)
target_link_options(glm53_verified_main_generic_regression PRIVATE "LINKER:--wrap=pread")
main_source_check(glm53_complete_main_source_check complete_reader_check.cpp)
