cmake_minimum_required(VERSION 3.24)
project(qfn8bit_hypothetical_fullgraph_plan LANGUAGES CXX)
find_package(Threads REQUIRED)
set(RESEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RADIANCE_SOURCE "${RESEARCH_ROOT}/radiance-src" CACHE PATH "Exact frozen stock Core")
set(RADIANCE_BUILD "${RESEARCH_ROOT}/build-ds41-aql" CACHE PATH "Matching parent-built whole AQL Core")
if(NOT EXISTS "${RADIANCE_BUILD}/core/librad_core.a")
  message(FATAL_ERROR "Supply matching existing whole AQL Core; no implicit build")
endif()
file(STRINGS "${RADIANCE_BUILD}/CMakeCache.txt" CORE_HOME REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" CORE_HOME "${CORE_HOME}")
file(REAL_PATH "${CORE_HOME}" ACTUAL_HOME)
file(REAL_PATH "${RADIANCE_SOURCE}" EXPECTED_HOME)
if(NOT ACTUAL_HOME STREQUAL EXPECTED_HOME)
  message(FATAL_ERROR "Private Core source/header/archive mismatch")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/CORE_SOURCE_PINS.json" PINS)
string(JSON COUNT LENGTH "${PINS}")
math(EXPR LAST "${COUNT}-1")
foreach(I RANGE 0 ${LAST})
  string(JSON REL MEMBER "${PINS}" ${I})
  string(JSON EXPECTED GET "${PINS}" "${REL}")
  file(SHA256 "${RADIANCE_SOURCE}/${REL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Frozen planning source changed: ${REL}")
  endif()
endforeach()
find_library(QFN_PLAN_HIP_LIBRARY amdhip64 REQUIRED)
find_library(QFN_PLAN_HSA_LIBRARY hsa-runtime64 REQUIRED)
add_executable(qfn8bit_plan qfn_plan.cpp)
target_compile_features(qfn8bit_plan PRIVATE cxx_std_20)
target_compile_definitions(qfn8bit_plan PRIVATE RAD_HAVE_HIP=1)
target_compile_options(qfn8bit_plan PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_include_directories(qfn8bit_plan PRIVATE "${RADIANCE_SOURCE}" "${RADIANCE_SOURCE}/core" "${RADIANCE_SOURCE}/abi"
  "${RADIANCE_SOURCE}/vendor" "${RADIANCE_BUILD}/rad_include")
target_link_libraries(qfn8bit_plan PRIVATE
  "-Wl,--whole-archive" "${RADIANCE_BUILD}/core/librad_core.a" "-Wl,--no-whole-archive"
  "${RADIANCE_BUILD}/vendor/librad_vendor_llama.a" "${RADIANCE_BUILD}/vendor/librad_vendor_httplib.a"
  ${QFN_PLAN_HIP_LIBRARY} ${QFN_PLAN_HSA_LIBRARY} Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(qfn8bit_plan PROPERTIES ENABLE_EXPORTS ON)
