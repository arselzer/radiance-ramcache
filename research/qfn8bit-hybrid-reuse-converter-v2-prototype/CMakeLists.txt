cmake_minimum_required(VERSION 3.24)
project(qfn8bit_hybrid_reuse_converter LANGUAGES CXX)
find_package(Threads REQUIRED)
set(RESEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RADIANCE_SOURCE "${RESEARCH_ROOT}/radiance-src" CACHE PATH "Exact frozen Core")
set(RADIANCE_BUILD "${RESEARCH_ROOT}/build-ds41-aql" CACHE PATH "Matching actual converter/Core build")
file(STRINGS "${RADIANCE_BUILD}/CMakeCache.txt" CORE_HOME REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" CORE_HOME "${CORE_HOME}")
file(REAL_PATH "${CORE_HOME}" ACTUAL_HOME)
file(REAL_PATH "${RADIANCE_SOURCE}" EXPECTED_HOME)
if(NOT ACTUAL_HOME STREQUAL EXPECTED_HOME)
  message(FATAL_ERROR "Private Core source/header/archive mismatch")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/CORE_SOURCE_PINS.json" PINS)
string(JSON COUNT LENGTH "${PINS}")
math(EXPR LAST "${COUNT}-1")
foreach(I RANGE 0 ${LAST})
  string(JSON REL MEMBER "${PINS}" ${I})
  string(JSON EXPECTED GET "${PINS}" "${REL}")
  file(SHA256 "${RADIANCE_SOURCE}/${REL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Frozen converter source changed: ${REL}")
  endif()
endforeach()
add_library(qfn_alias_arch MODULE alias_arch.cpp)
add_library(qfn_alias_layout MODULE "${RADIANCE_SOURCE}/core/plugin/testplugin/importkernel.cc")
foreach(TARGET qfn_alias_arch qfn_alias_layout)
  target_compile_features(${TARGET} PRIVATE cxx_std_20)
  target_include_directories(${TARGET} PRIVATE "${RADIANCE_SOURCE}/abi")
  target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endforeach()
find_library(QFN_CONVERT_HIP_LIBRARY amdhip64 REQUIRED)
find_library(QFN_CONVERT_HSA_LIBRARY hsa-runtime64 REQUIRED)
add_executable(qfn_alias_converter_check checker.cpp)
target_compile_features(qfn_alias_converter_check PRIVATE cxx_std_20)
target_compile_definitions(qfn_alias_converter_check PRIVATE RAD_HAVE_HIP=1)
target_compile_options(qfn_alias_converter_check PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_include_directories(qfn_alias_converter_check PRIVATE "${RADIANCE_SOURCE}" "${RADIANCE_SOURCE}/core" "${RADIANCE_SOURCE}/abi" "${RADIANCE_SOURCE}/vendor" "${RADIANCE_BUILD}/rad_include")
target_link_libraries(qfn_alias_converter_check PRIVATE
  "-Wl,--whole-archive" "${RADIANCE_BUILD}/core/librad_core.a" "-Wl,--no-whole-archive"
  "${RADIANCE_BUILD}/vendor/librad_vendor_llama.a" "${RADIANCE_BUILD}/vendor/librad_vendor_httplib.a"
  ${QFN_CONVERT_HIP_LIBRARY} ${QFN_CONVERT_HSA_LIBRARY} Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(qfn_alias_converter_check PROPERTIES ENABLE_EXPORTS ON)
add_dependencies(qfn_alias_converter_check qfn_alias_arch qfn_alias_layout)
# Manual HOST-only process, no CTest/inference/serving or implicit run.
